% =======================================================================
% ZJU LaTeX Resume Template —— 骨架文件
%
% 正文不在这里，而由「岗位清单」profiles/<jobname>.tex 决定。
% 清单文件名必须与 -jobname 一致：
%
%   latexmk                     -> profiles/CV.tex        -> CV.pdf        （默认/通用）
%   latexmk -jobname=CV-agent   -> profiles/CV-agent.tex  -> CV-agent.pdf  （Agent 开发）
%   latexmk -jobname=CV-neural  -> profiles/CV-neural.tex -> CV-neural.pdf （神经信息解码）
%
% 加新岗位：复制一份 profiles/*.tex 改成自己的内容即可，本文件不用动。
% 注意：字体/图片/清单的路径都相对「当前工作目录」解析，请在项目根目录编译。
% =======================================================================

\documentclass[10pt]{article}

\input{preamble/style}

\begin{document}
\pagenumbering{gobble}

\IfFileExists{profiles/\jobname.tex}
  {\input{profiles/\jobname.tex}}
  {\GenericError{}
    {找不到岗位清单 profiles/\jobname.tex}
    {zju-resume-template}
    {用 -jobname=CV-xxx 指定岗位，或直接 latexmk 编译默认版本。清单文件需放在 profiles/ 下。}}

\end{document}
